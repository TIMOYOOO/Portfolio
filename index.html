--- Crée un propriétaire de ressources, libérable indépendamment.
function Scope.New(name, report)
    return setmetatable({name = name, resources = {}, sequence = 0, stopped = false, report = report}, Scope)
end

--- Enregistre un nettoyage et renvoie une fonction qui le détache sans l'exécuter.
function Scope:Defer(callback)
    assert(type(callback) == "function", "cleanup_function_required")
    if self.stopped then callback(); return function() end end
    self.sequence = self.sequence + 1
    local id = self.sequence
    self.resources[id] = callback
    return function() self.resources[id] = nil end
end

--- Libère toutes les ressources une seule fois, dans l'ordre inverse d'acquisition.
function Scope:Stop()
    if self.stopped then return end
    self.stopped = true
    -- … nettoyages exécutés du plus récent au plus ancien
end

Projet Unreal II

Framework de gameplay maison. Chaque package crée son propre runtime, composé de services indépendants : actions, capacités, dégâts, buffs, réseau, synchronisation. Les capacités sont déclaratives : phases, file d'action et règles d'annulation sont validées dès l'enregistrement.

Voir le code 
--- Crée un runtime propre au package qui l'utilise, jamais un singleton partagé.
return function(engine)
    local fw = {Engine = engine, Scope = Scope}
    fw.Lifecycle = Scope.New("runtime", engine.Report)
    fw.StateMachine = makeMachines(fw)
    fw.Events       = makeEvents(fw)
    fw.Net          = makeNet(fw)
    fw.Sync         = makeSync(fw)
    fw.Actions      = makeActions(fw)
    fw.Ability      = makeAbilities(fw)
    fw.Buffs        = makeBuffs(fw)
    fw.Damage       = makeDamage(fw)
    -- … autres services
    return fw
end
--- Enregistre une capacité : décharger son contenu annule ses lancements et ses recharges.
function service.Register(id, definition, scope)
    assert(fw.Engine.IsServer and not fw.Lifecycle.stopped, "server_only")
    assert(type(definition.execute) == "function", "execute_required")
    local def = {}
    -- Chaque capacité occupe une file : sort, corps à corps ou exclusive
    def.lane = definition.lane or "exclusive"
    assert(def.lane == "spell" or def.lane == "melee" or def.lane == "exclusive", "invalid_ability_lane")
    -- Chaque phase a une durée bornée, vérifiée dès l'enregistrement
    for _, phase in ipairs({"preparation", "execution", "channel", "recovery", "cooldown"}) do
        def[phase] = definition[phase] == nil and 0 or definition[phase]
        assert(duration(def[phase]), "invalid_" .. phase)
    end
    -- Par défaut, on peut annuler pendant la préparation et la canalisation
    def.cancelable = {preparing = true, channeling = true}
    -- … callbacks et règles d'annulation
end

TMY-03 / CityEn cours

Cyber City-builder

Jeu de construction de ville dans un univers cyberpunk. Les effets des bâtiments reposent sur un système modulaire : chaque capacité associe un déclencheur et un effet, tous deux bâtis sur des classes abstraites. Un nouvel effet s'ajoute en une classe, puis s'assemble dans l'éditeur Unity sans toucher au reste du code.

Rendu Ambiance néon nocturne construite en post-process : le bloom fait rayonner les fenêtres et les enseignes émissives, la profondeur de champ et la vignette concentrent le regard sur la ville, le grain de film donne la texture finale.

public abstract class AbilityTrigger : ScriptableObject
{
    public abstract void InitAbility(AbilityContext abilityContext, AbilityEffect abilityEffect);
    public abstract void CleanupAbility(AbilityContext abilityContext, AbilityEffect abilityEffect);
}

public abstract class AbilityEffect : ScriptableObject
{
    public abstract void ApplyAbility(AbilityContext abilityContext);
}

// Une capacité = un déclencheur + un effet, assemblés dans l'inspecteur
[Serializable]
public class AbilityV2
{
    public AbilityTrigger abilityTrigger;
    public AbilityEffect abilityEffect;

    public void InitAbility(AbilityContext abilityContext)
    {
        abilityTrigger.InitAbility(abilityContext, abilityEffect);
    }
}

// Exemple d'effet : +N ressources par voisin du même type
public class GainResourcePerAdjacentSameType : AbilityEffect
{
    public ResourceType resourceType = ResourceType.VictoryPoints;
    [Min(0)] public int quantityPerNeighbor = 1;

    public override void ApplyAbility(AbilityContext abilityContext)
    {
        BuildingInstance own = abilityContext.Owner.GetComponent<BuildingInstance>();
        int count = 0;
        foreach (GameObject neighbor in GridManager.Instance.GetOrthogonalAdjacentBuildings(abilityContext.Owner))
            if (neighbor.GetComponent<BuildingInstance>()?.Data == own.Data) count++;

        if (count > 0)
            ResourceManager.Instance.AddResource(resourceType, count * quantityPerNeighbor);
    }
}
