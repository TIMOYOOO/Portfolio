--- Enregistre une capacité : décharger son contenu annule ses lancements et ses recharges.
function service.Register(id, definition, scope)
    assert(fw.Engine.IsServer and not fw.Lifecycle.stopped, "server_only")
    assert(type(definition.execute) == "function", "execute_required")
    local def = {}
    -- Chaque capacité occupe une file : sort, corps à corps ou exclusive
    def.lane = definition.lane or "exclusive"
    assert(def.lane == "spell" or def.lane == "melee" or def.lane == "exclusive", "invalid_ability_lane")
    -- Chaque phase a une durée bornée, vérifiée dès l'enregistrement
    for _, phase in ipairs({"preparation", "execution", "channel", "recovery", "cooldown"}) do
        def[phase] = definition[phase] == nil and 0 or definition[phase]
        assert(duration(def[phase]), "invalid_" .. phase)
    end
    -- Par défaut, on peut annuler pendant la préparation et la canalisation
    def.cancelable = {preparing = true, channeling = true}
    -- … callbacks et règles d'annulation
end

Projet Source 2

Gamemode multijoueur en développement. Les quêtes sont assemblées dans l'éditeur à partir de modules indépendants (déclencheurs, conditions, objectifs, récompenses, conséquences), sérialisées en JSON et validées avant d'être jouées. // Chaque brique de quête est un module abstrait : on en crée un nouveau en une classe
public abstract class QuestTrigger : QuestModule
{
    public abstract bool Matches( IQuestEvent ev, QuestContext ctx );
}

public abstract class QuestCondition : QuestModule
{
    public abstract bool IsMet( QuestContext ctx );
    // Réévaluée seulement quand un événement peut la faire changer
    public virtual bool IsInvalidatedBy( IQuestEvent ev ) => true;
}

public abstract class QuestObjective : QuestModule
{
    public abstract void OnActivated( QuestInstance inst, QuestContext ctx );
    public abstract void HandleEvent( QuestInstance inst, QuestContext ctx, IQuestEvent ev );
    public abstract ObjectiveStatus Evaluate( QuestInstance inst );
}

public abstract class QuestReward : QuestModule
{
    public abstract void Grant( QuestContext ctx );
}

public abstract class QuestConsequence : QuestModule
{
    public abstract void Execute( QuestContext ctx );
}
// Exemple d'objectif : atteindre une zone, configurable depuis l'éditeur
[Title( "Reach a location" ), Icon( "place" ), Category( "Objectives" )]
public class ReachLocationObjective : QuestObjective
{
    [Property, Title( "Zone Id" )]
    public string ZoneId { get; set; } = "";

    public override void OnActivated( QuestInstance inst, QuestContext ctx )
        => inst.StateOf( this ).Status = ObjectiveStatus.Active;

    public override void HandleEvent( QuestInstance inst, QuestContext ctx, IQuestEvent ev )
    {
        var state = inst.StateOf( this );
        if ( state.Status != ObjectiveStatus.Active )
            return;

        if ( ev is ZoneEnteredEvent zone && zone.ZoneId == ZoneId )
            state.Status = ObjectiveStatus.Completed;
    }

    public override ObjectiveStatus Evaluate( QuestInstance inst ) => inst.StateOf( this ).Status;

    // Une quête mal configurée est signalée dans l'éditeur, pas en jeu
    public override IEnumerable<QuestValidationError> Validate( QuestResource quest )
    {
        if ( string.IsNullOrWhiteSpace( ZoneId ) )
            yield return new QuestValidationError( this, "Objective \"Reach a location\" has no zone Id." );
    }
}

// Exemple d'effet : +N ressources par voisin du même type
public class GainResourcePerAdjacentSameType : AbilityEffect
{
    public ResourceType resourceType = ResourceType.VictoryPoints;
    [Min(0)] public int quantityPerNeighbor = 1;

    public override void ApplyAbility(AbilityContext abilityContext)
    {
        BuildingInstance own = abilityContext.Owner.GetComponent<BuildingInstance>();
        int count = 0;
        foreach (GameObject neighbor in GridManager.Instance.GetOrthogonalAdjacentBuildings(abilityContext.Owner))
            if (neighbor.GetComponent<BuildingInstance>()?.Data == own.Data) count++;

        if (count > 0)
            ResourceManager.Instance.AddResource(resourceType, count * quantityPerNeighbor);
    }
}
